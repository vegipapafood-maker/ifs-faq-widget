// Vercel Serverless Function: /api/widget // Supports Vercel KV Storage, Upstash Redis, and In-Memory Fallback interface WidgetRequestBody { id?: string; config: any; } // Global in-memory storage fallback for serverless warm instances const inMemoryStore = new Map(); function generateId(length: number = 7): string { const chars = 'abcdefghjkmnpqrstuvwxyz23456789'; let result = ''; for (let i = 0; i < length; i++) { result += chars.charAt(Math.floor(Math.random() * chars.length)); } return result; } export default async function handler(req: any, res: any) { // Set CORS headers so Canva or external embeds can freely query the API res.setHeader('Access-Control-Allow-Origin', '*'); res.setHeader('Access-Control-Allow-Methods', 'GET, POST, OPTIONS'); res.setHeader('Access-Control-Allow-Headers', 'Content-Type, Authorization'); if (req.method === 'OPTIONS') { return res.status(200).end(); } const kvUrl = process.env.KV_REST_API_URL || process.env.UPSTASH_REDIS_REST_URL; const kvToken = process.env.KV_REST_API_TOKEN || process.env.UPSTASH_REDIS_REST_TOKEN; // POST: Save widget configuration to database if (req.method === 'POST') { try { let body: WidgetRequestBody; if (typeof req.body === 'string') { body = JSON.parse(req.body); } else { body = req.body || {}; } if (!body.config || !Array.isArray(body.config.items)) { return res.status(400).json({ error: 'Invalid configuration: items array required' }); } const id = (body.id && typeof body.id === 'string' && body.id.trim()) ? body.id.trim() : generateId(7); // Save to Vercel KV if environment variables are set if (kvUrl && kvToken) { try { const kvRes = await fetch(`${kvUrl}/set/faq_${id}`, { method: 'POST', headers: { Authorization: `Bearer ${kvToken}`, 'Content-Type': 'application/json', }, body: JSON.stringify(body.config), }); if (kvRes.ok) { return res.status(200).json({ success: true, id, provider: 'vercel-kv', url: `https://ifs-faq-widget.vercel.app/?id=${id}&mode=embed`, }); } } catch (kvErr) { console.error('Vercel KV write error, falling back to memory/cloud:', kvErr); } } // Memory Store fallback inMemoryStore.set(`faq_${id}`, body.config); return res.status(200).json({ success: true, id, provider: 'api', url: `https://ifs-faq-widget.vercel.app/?id=${id}&mode=embed`, }); } catch (err: any) { return res.status(500).json({ error: err.message || 'Failed to save widget' }); } } // GET: Retrieve widget configuration by ID if (req.method === 'GET') { try { let id = req.query && typeof req.query.id === 'string' ? req.query.id : null; if (!id) { try { const url = new URL(req.url, 'http://localhost:3000'); id = url.searchParams.get('id'); } catch { // ignore } } if (!id) { return res.status(400).json({ error: 'Missing widget ID' }); } // 1. Try Vercel KV if (kvUrl && kvToken) { try { const kvRes = await fetch(`${kvUrl}/get/faq_${id}`, { headers: { Authorization: `Bearer ${kvToken}` }, }); if (kvRes.ok) { const data = await kvRes.json(); if (data && data.result) { const parsed = typeof data.result === 'string' ? JSON.parse(data.result) : data.result; return res.status(200).json({ success: true, config: parsed, provider: 'vercel-kv' }); } } } catch (kvErr) { console.error('Vercel KV read error:', kvErr); } } // 2. Try In-Memory Store if (inMemoryStore.has(`faq_${id}`)) { const cached = inMemoryStore.get(`faq_${id}`); return res.status(200).json({ success: true, config: cached, provider: 'api' }); } return res.status(404).json({ error: 'Widget configuration not found' }); } catch (err: any) { return res.status(500).json({ error: err.message || 'Failed to retrieve widget' }); } } return res.status(405).json({ error: 'Method not allowed' }); }
